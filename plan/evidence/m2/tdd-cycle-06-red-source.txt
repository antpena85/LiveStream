package livestream.feasibility

import java.nio.file.Path
import java.sql.DriverManager

data class ProbeSnapshot(val points: Int, val revision: Long, val epoch: Long = 0)

/** Isolated SQLite feasibility fixture, not the application's match schema. */
class DurableCommandProbe(
    database: Path,
    private val beforeCommit: () -> Unit = {},
) : AutoCloseable {
    private val connection = DriverManager.getConnection("jdbc:sqlite:${database.toAbsolutePath()}")

    init {
        connection.createStatement().use { statement ->
            statement.execute("PRAGMA synchronous=FULL")
            statement.execute("CREATE TABLE IF NOT EXISTS state (id INTEGER PRIMARY KEY CHECK(id=1), points INTEGER NOT NULL, revision INTEGER NOT NULL)")
            statement.execute("CREATE TABLE IF NOT EXISTS receipts (command_id TEXT PRIMARY KEY, points INTEGER NOT NULL, revision INTEGER NOT NULL)")
            statement.execute("INSERT OR IGNORE INTO state VALUES (1, 7, 0)")
        }
    }

    fun snapshot(): ProbeSnapshot = connection.createStatement().use { statement ->
        statement.executeQuery("SELECT points, revision FROM state WHERE id=1").use { rows ->
            check(rows.next())
            ProbeSnapshot(rows.getInt(1), rows.getLong(2))
        }
    }

    @Synchronized
    fun reset(confirmed: Boolean, expectedEpoch: Long): ProbeSnapshot = snapshot()

    @Synchronized
    fun recordPoint(commandId: String, expectedEpoch: Long): ProbeSnapshot {
        connection.autoCommit = false
        try {
            val existing = connection.prepareStatement("SELECT points, revision FROM receipts WHERE command_id=?").use { statement ->
                statement.setString(1, commandId)
                statement.executeQuery().use { rows ->
                    if (rows.next()) ProbeSnapshot(rows.getInt(1), rows.getLong(2)) else null
                }
            }
            if (existing != null) {
                connection.commit()
                return existing
            }
            connection.createStatement().use { statement ->
                statement.executeUpdate("UPDATE state SET points=points+1, revision=revision+1 WHERE id=1")
            }
            val accepted = snapshot()
            connection.prepareStatement("INSERT INTO receipts VALUES (?, ?, ?)").use { statement ->
                statement.setString(1, commandId)
                statement.setInt(2, accepted.points)
                statement.setLong(3, accepted.revision)
                statement.executeUpdate()
            }
            beforeCommit()
            connection.commit()
            return accepted
        } catch (failure: Exception) {
            connection.rollback()
            throw failure
        } finally {
            connection.autoCommit = true
        }
    }

    override fun close() = connection.close()
}
